# ==============================================================================
# SECTION 1: TRANSFER LOCAL FILE TO REMOTE SERVER (LOCAL -> REMOTE)
# ==============================================================================

# Syntax: scp <local_file> <username>@<remote_ip>:<remote_directory>
scp file1.txt paul@10.21.55.11:/tmp/

# Transfer multiple specific files at once to remote server:
scp file1.txt file2.txt paul@10.21.55.11:/tmp/

# Transfer files using wildcards (e.g., all files starting with 'file'):
scp file* paul@10.21.55.11:/tmp/


# ==============================================================================
# SECTION 2: TRANSFER ENTIRE DIRECTORY TO REMOTE SERVER (RECURSIVE)
# ==============================================================================

# Use -r flag to copy a directory and all subfiles recursively:
scp -r ./scp_demo paul@10.21.55.11:/tmp/


# ==============================================================================
# SECTION 3: DOWNLOAD FILE FROM REMOTE SERVER TO LOCAL MACHINE (REMOTE -> LOCAL)
# ==============================================================================

# Syntax: scp <username>@<remote_ip>:<remote_file_path> <local_destination>
# Copy remote file to current working directory (indicated by '.'):
scp paul@10.21.55.11:/tmp/remote_file .

# Copy remote file to a specific local folder path:
scp paul@10.21.55.11:/tmp/remote_file /home/paul/downloads/


# ==============================================================================
# SECTION 4: ADVANCED OPTIONS (VERBOSE MODE & CUSTOM SSH PORTS)
# ==============================================================================

# Use -v flag for verbose/debugging output:
scp -v paul@10.21.55.11:/tmp/remote_file .

# Transfer file using a custom SSH port (e.g., Port 2222):
scp -P 2222 file1.txt paul@10.21.55.11:/tmp/
